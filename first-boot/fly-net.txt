# Fly Debian — first-boot Wi-Fi
#
# Copy this file onto the FAT volume labelled FLY-SETUP (next to README.txt)
# BEFORE the first power-on. Or run:
#
#   ./scripts/prepare-sd.sh /media/you/FLY-SETUP
#
# 2.4 GHz only on the Lite 2.1 onboard radio.
# Password is stored in plaintext on this volume.
# Leave WIFI_SSID=YourNetwork to skip auto-join and still run first-run setup.

WIFI_ENABLED=1
WIFI_SSID=YourNetwork
WIFI_PSK=YourPassword
WIFI_COUNTRY=US

# Optional static IPv4. Leave USE_STATIC=0 for DHCP.
USE_STATIC=0
STATIC_IP=192.168.1.50
STATIC_MASK=255.255.255.0
STATIC_GATEWAY=192.168.1.1
STATIC_DNS=1.1.1.1 8.8.8.8
